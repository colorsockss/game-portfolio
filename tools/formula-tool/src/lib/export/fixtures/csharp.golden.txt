using System;
using AbilityType = Game.Data.AbilityType;

namespace Game.Rules
{
    /// <summary>
    /// 공식이 능력치를 읽는 개체.
    /// </summary>
    public interface IAbilityOwner
    {
        double Get(AbilityType type);
    }

    /// <summary>
    /// 공식을 부르는 쪽이 넘기는 입력. 공식이 읽는 입력을 넘기지 않으면 던진다.
    /// </summary>
    public sealed class FormulaInput
    {
        /// <summary>
        /// 스킬 계수
        /// </summary>
        public double? Coefficient { get; set; }
    }

    /// <summary>
    /// 공식. 난수는 넘겨받은 random에서만 뽑는다.
    /// </summary>
    public static class Formulas
    {
        /// <summary>
        /// 치명타 확률
        /// </summary>
        private const double CritRate = 0.05;

        /// <summary>
        /// 피해 &lt;기본&gt;
        /// </summary>
        public static double Damage(IAbilityOwner caster, IAbilityOwner target, FormulaInput input, Random random) =>
            (Math.Max(1.0, ((caster.Get(AbilityType.Attack) * Given(input.Coefficient, "coefficient")) - target.Get(AbilityType.Defense))) * Rand(random, 0.9, 1.1));

        public static double Critical(IAbilityOwner caster, IAbilityOwner target, FormulaInput input, Random random) =>
            ((Rand(random) < CritRate) ? 1.5 : 1.0);

        public static double Total(IAbilityOwner caster, IAbilityOwner target, FormulaInput input, Random random) =>
            (Math.Floor(Clamp((Damage(caster, target, input, random) * Critical(caster, target, input, random)), 0.0, 100000.0) + 0.5) + (RandInt(random, 0.0, 2.0) * 0.0));

        public static double Halved(IAbilityOwner caster, IAbilityOwner target, FormulaInput input, Random random) =>
            ((((((-(Math.Min(Math.Min(caster.Get(AbilityType.Attack), 3.0), 2.0) / 2.0)) % 7.0) + Math.Pow(2.0, 3.0)) + Math.Floor(1.5)) + Math.Ceiling(Math.Abs((-0.5)))) + Math.Sqrt(4.0));

        private static double Given(double? value, string key) =>
            value ?? throw new ArgumentException($"Formula input '{key}' was not given.", nameof(value));

        private static double Clamp(double x, double lo, double hi) => Math.Min(Math.Max(x, lo), hi);

        // rand() → [0,1), rand(a) → [0,a), rand(a,b) → [a,b). 툴의 시뮬레이션과 같다.
        private static double Rand(Random random) => random.NextDouble();
        private static double Rand(Random random, double a) => random.NextDouble() * a;
        private static double Rand(Random random, double a, double b) => a + (random.NextDouble() * (b - a));

        private static double RandInt(Random random, double lo, double hi)
        {
            double a = Math.Floor(lo);
            return a + Math.Floor(random.NextDouble() * (Math.Floor(hi) - a + 1));
        }
    }
}
